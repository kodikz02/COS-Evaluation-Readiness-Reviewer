    function triggerSubmitConfirmation() {
      const unanswered = state.userAnswers.filter(a => a === null).length;

    // =========================================================
    // SCORE CALCULATION & RESULTS SCREEN

    function finalizeExam(isAutoTimeout) {
      if (state.isSubmitted) return;
      state.isSubmitted = true;
      clearInterval(state.timerInterval);

      let part1Correct = 0;
      let part2Correct = 0;

      state.questions.forEach((q, idx) => {
        if (state.userAnswers[idx] === q.answer) {
          if (idx < 5) part1Correct++;
          else part2Correct++;
        }
      });

      const totalCorrect = part1Correct + part2Correct;
      const totalItems = state.questions.length;
      const percentage = Math.round((totalCorrect / totalItems) * 100);
      const chosenTrack = TRACKS.find(t => t.id === state.selectedTrackId);

      // Populate Summary Metadata
      document.getElementById("resExamineeName").textContent = state.examineeName;
      document.getElementById("resExamineeId").textContent = state.examineeId;
      document.getElementById("resTrackName").textContent = chosenTrack.title;

      const now = new Date();
      document.getElementById("resTimestamp").textContent = now.toLocaleString("en-PH", {
        dateStyle: "medium",
        timeStyle: "short"
      }) + (isAutoTimeout ? " (Auto-Submitted on Timeout)" : "");

      document.getElementById("resPercentage").textContent = `${percentage}%`;
      document.getElementById("resTotalScore").textContent = totalCorrect;
      document.getElementById("resPart1Score").textContent = `${part1Correct} / 5 (${Math.round((part1Correct / 5) * 100)}%)`;
      document.getElementById("resPart2Score").textContent = `${part2Correct} / 10 (${Math.round((part2Correct / 10) * 100)}%)`;

      // Readiness Badge
      const badge = document.getElementById("readinessBadge");
      if (percentage >= 85) {
        badge.textContent = "Highly Ready (Exemplary)";
        badge.className = "inline-flex items-center gap-2 px-3.5 py-1.5 rounded-full text-xs font-extrabold uppercase tracking-wider bg-emerald-100 text-emerald-800";
      } else if (percentage >= 75) {
        badge.textContent = "Ready / Passing Competency";

      } else {
        badge.textContent = "Needs Technical Review (< 75%)";
        badge.className = "inline-flex items-center gap-2 px-3.5 py-1.5 rounded-full text-xs font-extrabold uppercase tracking-wider bg-rose-100 text-rose-800";

      renderDetailedReview();

      // Switch Screen
      examScreen.classList.add("hidden");
      resultsScreen.classList.remove("hidden");
      window.scrollTo({ top: 0, behavior: "smooth" });

    function renderDetailedReview() {
      const container = document.getElementById("reviewListContainer");
      container.innerHTML = "";
      const letters = ["A", "B", "C", "D"];

      state.questions.forEach((q, idx) => {
        const userChoice = state.userAnswers[idx];
        const isCorrect = userChoice === q.answer;

        const card = document.createElement("div");
        card.className = `bg-white rounded-2xl p-6 shadow-sm border-l-4 print-break-inside ${
          isCorrect ? "border-l-emerald-500 border border-slate-200" : "border-l-rose-500 border border-slate-200"
        }`;

        const statusTag = isCorrect
          ? `Correct`
          : `${userChoice === null ? "Unanswered" : "Incorrect"}`;

        let optionsHTML = "";
        q.options.forEach((opt, oIdx) => {
          const isRightAnswer = oIdx === q.answer;
          const isUserPicked = oIdx === userChoice;

          let rowStyle = "bg-slate-50 border-slate-200 text-slate-700";
          let marker = "";

          if (isRightAnswer) {
            rowStyle = "bg-emerald-50/90 border-emerald-400 text-emerald-950 font-semibold";
            marker = `✓ Correct Answer`;
          } else if (isUserPicked && !isRightAnswer) {
            rowStyle = "bg-rose-50/90 border-rose-400 text-rose-950 font-semibold";
            marker = `✗ Your Answer`;
          }

          optionsHTML += `
            

              ${letters[oIdx]}.
              ${escapeHTML(opt)}
              ${marker}
            

          `;
        });

        card.innerHTML = `
          

            

              Item #${idx + 1}
              •
              ${q.part}
            

            ${statusTag}
          


          
${escapeHTML(q.question)}


          ${q.snippet ? `
${escapeHTML(q.snippet)}
` : ""}

          

            ${optionsHTML}
          


          

            Technical Explanation: ${escapeHTML(q.explanation)}
          

        `;

        container.appendChild(card);
      });
    }
